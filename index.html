🏭 Pengantar Factory Constructors

Slide 2
  • Apa itu? Constructor khusus yang tidak selalu membuat instance baru.
  • Keyword: factory
  • Perbedaan dengan generative constructor:
    • Generative: Selalu membuat instance baru.
    • Factory: Bisa mengembalikan instance yang sudah ada, atau instance subclass.

🔧 Sintaks Factory Constructor

Slide 3
class MyClass {
  // Generative constructor (biasa)
  MyClass();

  // Factory constructor
  factory MyClass.create() {
    // Bisa return instance yang sudah ada
    // Atau instance dari subclass
    // Atau instance baru
    return MyClass();
  }
}

void main(){
  var obj1 = MyClass();      // Generative
  var obj2 = MyClass.create(); // Factory
}

🎯 Kasus 1 – Singleton Pattern

Slide 4

Singleton: Hanya satu instance sepanjang aplikasi.

class DatabaseConnection {
  // Private constructor
  DatabaseConnection._internal() {
    print('Creating database connection...');
  }
  // Static instance
  static final DatabaseConnection _instance = DatabaseConnection._internal();
  
  // Factory constructor
  factory DatabaseConnection() {
    return _instance;
  }
  
  void connect() {
    print('Connected to database');
  }
}

void main() {
  var db1 = DatabaseConnection();
  var db2 = DatabaseConnection();
  print(identical(db1, db2)); // true - instance yang sama
  db1.connect();
  db2.connect();
}

🗄️ Kasus 2 – Caching / Connection Pool

Slide 5
class ConnectionPool {
  final String host;
  final int port;
  // Cache berdasarkan parameter
  static final Map<String, ConnectionPool> _cache = {};

  // Private constructor
  ConnectionPool._(this.host, this.port);

  // Factory dengan caching
  factory ConnectionPool(String host, int port) {
    var key = '$host:$port';
    if (_cache.containsKey(key)) {
      print('Returning cache connection for $key');
      return _cache[key]!;
    }
    print('Creating new connection for $key');
    var connection = ConnectionPool._(host, port);
    _cache[key] = connection;
    return connection;
  }

  void query(String sql) {
    print('Executing: "$sql" on $host:$port');
  }
}

void main() {
  var conn1 = ConnectionPool('localhost', 5432);
  var conn2 = ConnectionPool('localhost', 5432); // dari cache
  var conn3 = ConnectionPool('db.example.com', 3306);
  conn1.query('SELECT * FROM users;');
  conn2.query('INSERT INTO products (name) VALUES ("Laptop");');
  print(identical(conn1, conn2)); // true
}
✨ Output demonstrasi: koneksi yang sama untuk host:port yang sama, mengurangi overhead.

🔺 Kasus 3 – Mengembalikan Subclass (Shape Factory)

Slide 6
abstract class Shape {
  void draw();
  
  factory Shape(String type) {
    switch (type) {
      case 'circle':
        return Circle();
      case 'square':
        return Square();
      case 'triangle':
        return Triangle();
      case 'rectangle':
        return Rectangle();
      default:
        throw ArgumentError("Unknown shape: $type");
    }
  }
}

class Circle implements Shape {
  @override void draw() => print("Drawing circle");
}
class Square implements Shape {
  @override void draw() => print("Drawing square");
}
class Triangle implements Shape {
  @override void draw() => print("Drawing triangle");
}
class Rectangle implements Shape {
  @override void draw() => print("Drawing rectangle");
}

void main() {
  var circle = Shape('circle');
  var square = Shape('square');
  var rect = Shape('rectangle');
  circle.draw();
  square.draw();
  rect.draw();
  // Shape('pentagon'); // error
}

⚙️ Praktik 1 – Factory untuk Configuration

Slide 7

Materi praktik: membuat factory untuk menyediakan konfigurasi aplikasi dengan caching & pembacaan dari environment (ilustrasi).

class AppConfig {
  final String apiUrl;
  final int timeout;

  static final Map<String, AppConfig> _cache = {};

  AppConfig._(this.apiUrl, this.timeout);

  factory AppConfig.fromEnvironment(String env) {
    if (_cache.containsKey(env)) return _cache[env]!;
    // simulasi load config
    if (env == 'production') {
      var config = AppConfig._('https://api.example.com', 30);
      _cache[env] = config;
      return config;
    } else {
      var config = AppConfig._('https://dev.api.example.com', 60);
      _cache[env] = config;
      return config;
    }
  }

  @override
  String toString() => 'AppConfig(apiUrl: $apiUrl, timeout: $timeout)';
}

💳 Praktik 2 – Factory untuk Payment Methods

Slide 10 (materi praktik)
abstract class PaymentMethod {
  void pay(double amount);
  
  factory PaymentMethod(String type) {
    switch (type.toLowerCase()) {
      case 'creditcard':
        return CreditCardPayment();
      case 'paypal':
        return PayPalPayment();
      case 'banktransfer':
        return BankTransferPayment();
      case 'ewallet':
        return EWalletPayment();
      default:
        throw ArgumentError('Unknown payment method: $type');
    }
  }
}

class CreditCardPayment implements PaymentMethod {
  @override void pay(double amount) => print('Paying \$$amount with Credit Card');
}
class PayPalPayment implements PaymentMethod {
  @override void pay(double amount) => print('Paying \$$amount with PayPal');
}
class BankTransferPayment implements PaymentMethod {
  @override void pay(double amount) => print('Paying \$$amount with Bank Transfer');
}
class EWalletPayment implements PaymentMethod {
  @override void pay(double amount) => print('Paying \$$amount with E-Wallet');
}

void main() {
  var method = PaymentMethod('paypal');
  method.pay(99.99);
}

🔄 Factory Constructor vs Static Method

Slide 11 (perbandingan)
class Example {
  Example._() {} // private constructor

  // Konstruktor pabrik (Factory Constructor)
  factory Example.create() => Example._();

  // Metode statis (Static Method)
  static Example createStatic() => Example._();
}

void main() {
  var e1 = Example.create();       // konstruktor pabrik
  var e2 = Example.createStatic(); // metode statis
  print('e1: $e1, e2: $e2');
  print(identical(e1, e2) ? "sama" : "berbeda instance");
}

Keduanya dapat digunakan, namun factory constructor lebih ekspresif dalam konteks "constructor" dan bisa digunakan di new / konstanta.

🎨 Factory untuk Immutable Objects dengan Cache

Slide 12
class Color {
  final int red;
  final int green;
  final int blue;

  static final Map<String, Color> _cache = {};

  // Private constructor
  Color._(this.red, this.green, this.blue);

  // Factory dengan cache
  factory Color.rgb(int red, int green, int blue) {
    if (red < 0 || red > 255 || green < 0 || green > 255 || blue < 0 || blue > 255) {
      throw ArgumentError('RGB values must be 0-255');
    }
    var key = '$red,$green,$blue';
    if (!_cache.containsKey(key)) {
      _cache[key] = Color._(red, green, blue);
    }
    return _cache[key]!;
  }

  // Named factories untuk warna umum
  factory Color.red() => Color.rgb(255, 0, 0);
  factory Color.green() => Color.rgb(0, 255, 0);
  factory Color.blue() => Color.rgb(0, 0, 255);
  factory Color.white() => Color.rgb(255, 255, 255);
  factory Color.black() => Color.rgb(0, 0, 0);

  @override
  String toString() => 'Color($red, $green, $blue)';
}

void main() {
  var red1 = Color.red();
  var red2 = Color.rgb(255, 0, 0);
  print(identical(red1, red2)); // true - instance sama
  print(red1);
  print(red2);
}

✅ Ringkasan & Best Practices

Slide 15-16

🎯 Poin Penting Factory Constructor

  • Factory constructor: factory class.name() { ... }
  • Bisa mengembalikan: instance yang ada (cache), subclass, atau instance baru.
  • Use cases: singleton, caching, object pooling, subclass selection, immutable objects pooling.

📌 Best Practices:

  • Gunakan factory untuk: Object pooling/caching, Singleton pattern, Complex object creation, Subclass selection.
  • Gunakan generative constructor untuk: Simple object creation, initialization lists, required parameters.
  • Validasi: Lakukan validasi di factory sebelum membuat objek (contoh pada Color).
⚡ Factory constructor sangat berguna untuk mengendalikan pembuatan objek, menghemat memori, dan menyediakan API yang bersih.

  • Subclass selection berdasarkan input (mengembalikan Dog, Cat, atau Bird).
